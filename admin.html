const fileInput = document.getElementById("ad-file"); const file = fileInput.files[0]; const videoUrl = document.getElementById("ad-video").value.trim(); const youtubeId = extractYouTubeId(videoUrl); finishPublish(); return; } // Validation image const maxSize = 6 * 1024 * 1024; if (!file) throw new Error("Aucun fichier sélectionné."); if (!file.type.startsWith("image/")) throw new Error("Le fichier doit être une image."); const imageUrl = json.data.url; const deleteUrl = json.data.delete_url || json.data.delete || null; await db.collection("realisations_golden").add({ zone, title, description, link, imageUrl, videoId: null, imgbbDeleteUrl: deleteUrl, createdAt: firebase.firestore.FieldValue.serverTimestamp() });
