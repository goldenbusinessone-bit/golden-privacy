// =============================== // CONFIGURATION (clé cachée dans config.js) // =============================== const IMGBB_API_KEY = window.IMGBB_API_KEY; const ADMIN_UID = window.GOLDEN_ADMIN_UID; // =============================== // HELPERS // =============================== function extractYouTubeId(url){ const m = String(url || "").match(/(?:youtu\.be\/|youtube\.com\/(?:watch\?v=|shorts\/|embed\/))([A-Za-z0-9_-]{11})/); return m ? m[1] : null; } function showStatus(text, isError = false){ const status = document.getElementById("upload-status"); status.style.display = "block"; status.style.color = isError ? "#c85a5a" : "#b8860b"; status.textContent = text; } function clearStatus(){ const status = document.getElementById("upload-status"); status.style.display = "none"; status.textContent = ""; } // =============================== // AUTH ADMIN // =============================== async function adminSignIn(){ const email = document.getElementById("admin-email").value.trim(); err.style.display = "block"; return; } document.getElementById("btn-signin").disabled = true; try{ await auth.signInWithEmailAndPassword(email, pass); }catch(e){ err.textContent = e.message; err.style.display = "block"; document.getElementById("btn-signin").disabled = false; } } auth.onAuthStateChanged(user=>{ if(user && user.uid === ADMIN_UID){ document.getElementById("auth-card").style.display = "none"; document.getElementById("admin-panel").style.display = "block"; loadAdminList(); }else{ document.getElementById("auth-card").style.display = "block"; document.getElementById("admin-panel").style.display = "none"; document.getElementById("btn-signin").disabled = false; } }); // =============================== // PUBLISH REALISATION // =============================== async function publishRealisation(e){ const btn = document.getElementById("btn-publish"); btn.disabled = true; try{ const zone = document.getElementById("ad-zone").value; const title = document.getElementById("ad-title").value.trim(); const description = document.getElementById("ad-desc").value.trim(); const link = document.getElementById("ad-link").value.trim(); const file = document.getElementById("ad-file").files[0]; const videoUrl = document.getElementById("ad-video").value.trim(); const youtubeId = extractYouTubeId(videoUrl); if(!title || (!file && !youtubeId)){ alert("Ajoute un titre et une image ou une vidéo."); btn.disabled = false; return; } showStatus("Préparation..."); if(youtubeId){ showStatus("Publication vidéo..."); await db.collection("realisations_golden").add({ zone, title, description, link, imageUrl:`https://img.youtube.com/vi/${youtubeId}/hqdefault.jpg`, videoId:youtubeId, imgbbDeleteUrl:null, createdAt:firebase.firestore.FieldValue.serverTimestamp() }); finishPublish(); return; } const maxSize = 6 * 1024 * 1024; if(!file) throw new Error("Aucun fichier."); if(!file.type.startsWith("image/")) throw new Error("Fichier non image."); if(file.size > maxSize) throw new Error("Image > 6MB."); showStatus("Envoi ImgBB..."); const formData = new FormData(); formData.append("image", file); const res = await fetch("https://api.imgbb.com/1/upload?key=" + IMGBB_API_KEY, { method:"POST", body:formData }); const json = await res.json(); if(!json.success) throw new Error("Échec ImgBB"); const imageUrl = json.data.url; const deleteUrl = json.data.delete_url || json.data.delete || null; await db.collection("realisations_golden").add({ zone, title, description, link, imageUrl, videoId:null, imgbbDeleteUrl:deleteUrl, createdAt:firebase.firestore.FieldValue.serverTimestamp() }); finishPublish(); }catch(err){ showStatus("❌ " + err.message, true); btn.disabled = false; setTimeout(clearStatus, 3000); } } function finishPublish(){ showStatus("✅ Publié !"); document.getElementById("ad-title").value=""; document.getElementById("ad-desc").value=""; document.getElementById("ad-link").value=""; document.getElementById("ad-video").value=""; document.getElementById("ad-file").value=""; document.getElementById("btn-publish").disabled=false; loadAdminList(); setTimeout(clearStatus, 2000); } // =============================== // LOAD LIST // =============================== let unsubscribeList = null; function loadAdminList(){ const list = document.getElementById("admin-list"); if(unsubscribeList) unsubscribeList(); unsubscribeList = db.collection("realisations_golden") .orderBy("createdAt","desc") .onSnapshot(snapshot=>{ list.innerHTML = ""; if(snapshot.empty){ list.innerHTML = '

Aucune réalisation.

'; return; } snapshot.docs.forEach(doc=>{ const d = doc.data(); const item = document.createElement("div"); item.className = "admin-list-item"; const img = document.createElement("img"); img.src = d.imageUrl; const info = document.createElement("div"); info.className = "info"; info.innerHTML = `${d.title}
${d.zone}
`; const btn = document.createElement("button"); btn.textContent = "Supprimer"; btn.className = "btn secondary"; btn.onclick = ()=>deleteRealisation(doc.id); item.appendChild(img); item.appendChild(info); item.appendChild(btn); list.appendChild(item); }); }); } // =============================== // DELETE REALISATION // =============================== async function deleteRealisation(id){ if(!confirm("Supprimer ?")) return; try{ const docRef = db.collection("realisations_golden").doc(id); const doc = await docRef.get(); const data = doc.data(); if(data.imgbbDeleteUrl){ try{ await fetch(data.imgbbDeleteUrl); } catch(e){ console.warn("Erreur suppression ImgBB"); } } await docRef.delete(); }catch(err){ alert("Erreur suppression."); } }
